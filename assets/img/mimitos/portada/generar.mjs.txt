/* chapita del collar: corazón o QR */
function tag(cx, cy, r, kind = 'heart') {
  const ring = `<circle cx="${cx}" cy="${cy - r - 4}" r="5" fill="none" stroke="${INK}" stroke-width="2.5"/>`;
  const base = `<circle cx="${cx}" cy="${cy}" r="${r}" fill="${HEART}" stroke="${INK}" stroke-width="${sw}"/>`;
  if (kind === 'heart') return ring + base + heartPath(cx, cy + 1, r / 17, PAPER);
  // QR mínimo: cuadrado blanco con tres marcas y puntitos
  const q = r * 1.1, x = cx - q / 2, y = cy - q / 2, m = q / 7;
  const mark = (mx, my) => `<rect x="${mx}" y="${my}" width="${m * 2.4}" height="${m * 2.4}" rx="${m * .5}" fill="none" stroke="${INK}" stroke-width="${m * .55}"/>`;
  const dots = [[4.3, 1.2], [5.6, 2.4], [1.2, 4.4], [3.2, 3.4], [4.4, 4.6], [5.6, 5.6], [3.2, 5.6], [2.2, 3.1]].map(([a, b]) => `<rect x="${x + a * m}" y="${y + b * m}" width="${m * .9}" height="${m * .9}" rx="${m * .2}" fill="${INK}"/>`).join('');
  return ring + base + `<rect x="${x - m * .6}" y="${y - m * .6}" width="${q + m * 1.2}" height="${q + m * 1.2}" rx="${m}" fill="${PAPER}"/>` + mark(x + m * .2, y + m * .2) + mark(x + q - m * 2.6, y + m * .2) + mark(x + m * .2, y + q - m * 2.6) + dots;
}
